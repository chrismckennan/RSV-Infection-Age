#Simulate birth dates#
DOB.sim <- as.Date("2012-06-01") + runif(n = floor(n/2), min = 0, max = as.numeric(as.Date("2012-12-31")-as.Date("2012-06-01")))
DOB.sim <- c(DOB.sim, as.Date("2013-06-01") + runif(n = n-length(DOB.sim), min = 0, max = as.numeric(as.Date("2013-12-31")-as.Date("2013-06-01"))))
DOB.sim <- sort(as.Date(DOB.sim))

#Generate Covariates#
Cov.sim <- Cov.sim[order(runif(n)),]

#Compute probabilities#
Prob.sim <- PredictCondDensity(Lambda = Lambda, DOB = DOB.sim, a.max = 365, knots = Knots, degree = 3, Alpha = W.params.true, Cov = Cov.sim, Beta = beta.true, Mu = rep(0,length(beta.true)), Month.use = 1:12, Cond.Prob = F)

#Simulate data#

#Create dataset#

Solid black line: estimate Dashed lines: 95% confidence intervals Red line: simulated ground-truth weight function Note: the dashed lines are point-wise confidence intervals, not confidence bands. Therefore, the red line is not expected to lie within the dashed lines at every point.
